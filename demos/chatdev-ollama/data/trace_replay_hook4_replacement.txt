### Step 1: Designing the Core Classes and Functions

To build a desktop calculator app using Tkinter, we need several components:

1. **CalculatorApp**: The main application class that sets up the GUI.
2. **DisplayWidget**: A widget to show the current calculation and result.
3. **ButtonGrid**: A grid of buttons for digits, operations, and control functions.
4. **CalculatorLogic**: Handles the logic for performing calculations based on user input.

This module contains the main application class for the calculator.

import tkinter as tk
from display_widget import DisplayWidget
from button_grid import ButtonGrid
from calculator_logic import CalculatorLogic

class CalculatorApp:
    def __init__(self, root):

        self.root.title("Calculator")
        
        self.display = DisplayWidget(root)

#### display_widget.py

```python
# display_widget.py

'''

This module contains the widget for displaying the calculator's output.
'''

import tkinter as tk

class DisplayWidget(tk.Entry):
    def __init__(self, master, **kwargs):
        super().__init__(master, **kwargs)
        self.config(font=('Arial', 16), justify='right')
        self.pack(side=tk.TOP, fill=tk.BOTH, expand=True)

    def update_display(self, text):
        self.delete(0, tk.END)
        self.insert(tk.END, text)
```

#### button_grid.py

```python
# button_grid.py

'''

This module contains the grid of buttons for the calculator.
'''

import tkinter as tk
from calculator_logic import CalculatorLogic

class ButtonGrid(tk.Frame):
    def __init__(self, master, display, logic, **kwargs):
        super().__init__(master, **kwargs)
        self.display = display
        self.logic = logic
        
        button_texts = [
            '7', '8', '9', '/',
            '4', '5', '6', '*',
            '1', '2', '3', '-',
            '0', '.', '=', '+',
            'C', '⌫'
        ]
        
        row, col = 0, 0
        for text in button_texts:
            if text == '=':
                tk.Button(self, text=text, font=('Arial', 16), command=lambda: self.logic.calculate()).grid(row=row, column=col, sticky=tk.NSEW)
            elif text == 'C':
                tk.Button(self, text=text, font=('Arial', 16), command=lambda: self.display.update_display('0')).grid(row=row, column=col, sticky=tk.NSEW)
            elif text == '⌫':
                tk.Button(self, text=text, font=('Arial', 16), command=lambda: self.display.update_display(self.display.get()[:-1])).grid(row=row, column=col, sticky=tk.NSEW)
            else:
                tk.Button(self, text=text, font=('Arial', 16), command=lambda t=text: self.display.update_display(self.display.get() + t)).grid(row=row, column=col, sticky=tk.NSEW)
            
            col += 1
            if col > 3:
                col = 0
                row += 1
        
        for i in range(4):
            self.grid_rowconfigure(i, weight=1)
            self.grid_columnconfigure(i, weight=1)
```

#### calculator_logic.py

```python
# calculator_logic.py

'''

This module contains the logic for performing calculations.
'''

class CalculatorLogic:
    def __init__(self):
        self.current_input = ''
        self.operator = None
        self.first_operand = None

    def calculate(self):
        try:
            second_operand = float(self.current_input)
            if self.operator == '+':
                result = self.first_operand + second_operand
            elif self.operator == '/':
                if second_operand == 0:
                    return 'error'
                result = self.first_operand / second_operand
            else:
                return self.current_input
            
            self.display.update_display(str(result))
        except ValueError:
            self.display.update_display('error')

    def set_operator(self, operator):
        if self.current_input == '':
            return
        if self.first_operand is None:
            self.first_operand = float(self.current_input)
        else:
            self.calculate()
        self.operator = operator
        self.current_input = ''

    def append_digit(self, digit):
        self.current_input += digit

    def clear(self):
        self.current_input = ''
        self.operator = None
        self.first_operand = None
```
