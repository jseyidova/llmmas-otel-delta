   - `clear_display`: Clears the display.
   - `backspace`: Removes the last character from the display.

from calculator_ui import CalculatorApp

def main():
    root = tk.Tk()

#### calculator_ui.py
```python
import tkinter as tk

class CalculatorApp:
    def __init__(self, master):
        self.master = master
        self.master.title("Calculator")
        
        # Display widget
        self.display = tk.Entry(master, width=40, borderwidth=5)
        self.display.grid(row=0, column=0, columnspan=4, padx=10, pady=10)
        
        # Button layout
        buttons = [
            ('7', 1, 0), ('8', 1, 1), ('9', 1, 2), ('/', 1, 3),
            ('4', 2, 0), ('5', 2, 1), ('6', 2, 2), ('*', 2, 3),
            ('1', 3, 0), ('2', 3, 1), ('3', 3, 2), ('-', 3, 3),
            ('.', 4, 0), ('0', 4, 1), ('=', 4, 2), ('+', 4, 3),
            ('C', 5, 0), ('⌫', 5, 1)
        ]
        
        for (text, row, col) in buttons:
            button = tk.Button(master, text=text, width=10, height=3,
                               command=lambda t=text: self.on_button_click(t))
            button.grid(row=row, column=col, padx=5, pady=5)
    
    def on_button_click(self, char):
        if char == '=':
            self.evaluate_expression()
        elif char == '⌫':
            self.backspace()
        else:
            self.update_display(char)
    
    def update_display(self, char):
        current_text = self.display.get()
        new_text = current_text + char
        self.display.delete(0, tk.END)
        self.display.insert(0, new_text)
    
    def evaluate_expression(self):
        try:
            result = eval(self.display.get())
            self.display.delete(0, tk.END)
            self.display.insert(0, str(result))
        except Exception as e:
            self.display.delete(0, tk.END)
            self.display.insert(0, f"error: {str(e)}")
    
    def clear_display(self):
        self.display.delete(0, tk.END)
        self.display.insert(0, '0')
    
    def backspace(self):
        current_text = self.display.get()
        if len(current_text) > 1:
            new_text = current_text[:-1]
        else:
            new_text = '0'
        self.display.delete(0, tk.END)
        self.display.insert(0, new_text)
```

### Explanation

- **main.py**: This file initializes the Tkinter application and starts the main loop.
- **calculator_ui.py**: This file contains the `CalculatorApp` class which sets up the GUI and handles button clicks. The `on_button_click` method routes different button actions to appropriate methods (`update_display`, `evaluate_expression`, `clear_display`, `backspace`). The `evaluate_expression` method uses Python's `eval` function to compute the result of the expression, handling division by zero and other exceptions gracefully.

This setup ensures that all required functionalities are implemented and the application runs as expected with Python 3.10+.
